65fn compute_checksum(words: &[Word72]) -> u128 {
66    words.iter().fold(0u128, |acc, w| acc.wrapping_add(w.raw()))
67}
68
69fn check_drum_bounds(start: Address, len: usize) -> Result<usize, SpeedError> {
70    let start_idx = usize::from(start.raw());
71    if len == 0 {
72        return Ok(start_idx);
73    }
74    if start_idx.saturating_add(len) > DRUM_CAPACITY {
75        let end = u32::from(start.raw())
76            .saturating_add(u32::try_from(len).unwrap_or(u32::MAX))
77            .saturating_sub(1);
78        return Err(SpeedError::DrumAddressOutOfRange {
79            start: start.raw(),
80            end,
81        });
82    }
83    Ok(start_idx)
84}

86/// Simulated IBM 701 magnetic tape drive supporting SpeedCo I tape semantics.
87#[derive(Debug, Clone)]
88pub struct TapeUnit {
89    id: TapeId,
90    status: TapeStatus,
91    records: Vec<TapeRecord>,
92    position: usize,
93    rewound: bool,
94}
95
96impl TapeUnit {
97    /// Creates a new rewound tape unit in `Neutral` status.
98    pub fn new(id: TapeId) -> Self {
99        Self {
100            id,
101            status: TapeStatus::Neutral,
102            records: Vec::new(),
103            position: 0,
104            rewound: true,
105        }
106    }
107
108    /// Returns the tape unit identifier (`J`, `K`, `L`, or `M`).
109    pub fn id(&self) -> TapeId {
110        self.id
111    }
112
113    /// Returns current read/neutral/write status.
114    pub fn status(&self) -> TapeStatus {
115        self.status
116    }
117
118    /// Returns `true` when the tape is at the beginning-of-file / rewound position.
119    pub fn is_rewound(&self) -> bool {
120        self.rewound
121    }
122
123    /// Returns total number of unit records written in the current file.
124    pub fn record_count(&self) -> usize {
125        self.records.len()
126    }
127
128    /// Returns the current inter-record gap cursor index (`0` = start of file).
129    pub fn position(&self) -> usize {
130        self.position
131    }
132
133    /// Executes `WRTP*`: writes `words` as one unit record if status permits (`Neutral`/`Write`).
134    pub fn write_record(&mut self, words: &[Word72]) -> Result<Option<MachineStop>, SpeedError> {
135        if self.status == TapeStatus::Read {
136            return Ok(Some(MachineStop::CopyCheck));
137        }
138        self.records.truncate(self.position);
139        self.records.push(TapeRecord {
140            words: words.to_vec(),
141            checksum: compute_checksum(words),
142            parity_corrupted: false,
143            checksum_corrupted: false,
144        });
145        self.position = self.records.len();
146        self.status = TapeStatus::Write;
147        self.rewound = false;
148        Ok(None)
149    }
150
151    /// Executes `EFTP*`: marks end-of-file gap (transitions `Write -> Neutral`; ignored in `Read`).
152    pub fn end_file(&mut self) -> Result<(), SpeedError> {
153        if self.status == TapeStatus::Write {
154            self.status = TapeStatus::Neutral;
155            self.rewound = false;
156        }
157        Ok(())
158    }
159
160    /// Executes `RWTP*`: rewinds tape and restores `Neutral` status.
161    pub fn rewind(&mut self) {
162        self.position = 0;
163        self.status = TapeStatus::Neutral;
164        self.rewound = true;
165    }
166
167    /// Executes `RFTP*`: reads forward and returns up to `count` words (`B - A + 1`).
168    pub fn read_forward(&mut self, count: usize) -> Result<TapeReadOutcome, SpeedError> {
169        if self.status == TapeStatus::Write {
170            return Ok(TapeReadOutcome::Stop(MachineStop::CopyCheck));
171        }
172        self.status = TapeStatus::Read;
173        self.rewound = false;
174        if self.position >= self.records.len() {
175            return Ok(TapeReadOutcome::EndOfFileSkip);
176        }
177        let idx = self.position;
178        self.position += 1;
179        let rec = &self.records[idx];
180        if count > rec.words.len() {
181            return Ok(TapeReadOutcome::Stop(MachineStop::CopyCheck));
182        }
183        if rec.parity_corrupted {
184            return Ok(TapeReadOutcome::Stop(MachineStop::TapeCheck));
185        }
186        if rec.checksum_corrupted || compute_checksum(&rec.words) != rec.checksum {
187            return Ok(TapeReadOutcome::ChecksumMismatchSkip);
188        }
189        Ok(TapeReadOutcome::Ok(rec.words[..count].to_vec()))
190    }
191
192    /// Executes `RBTP*`: backspaces, reads forward, and backspaces again (except on error).
193    pub fn read_backward(&mut self, count: usize) -> Result<TapeReadOutcome, SpeedError> {
194        if self.status == TapeStatus::Write {
195            return Ok(TapeReadOutcome::Stop(MachineStop::CopyCheck));
196        }
197        self.status = TapeStatus::Read;
198        self.rewound = false;
199        if self.position == 0 {
200            return Ok(TapeReadOutcome::EndOfFileSkip);
201        }
202        self.position -= 1;
203        let outcome = self.read_forward(count)?;
204        if matches!(outcome, TapeReadOutcome::Ok(_)) {
205            self.position -= 1;
206        }
207        Ok(outcome)
208    }
209
210    /// Executes `SFTP*`: advances over the next unit record without reading.
211    pub fn skip_forward(&mut self) -> Result<TapeSkipOutcome, SpeedError> {
212        if self.status == TapeStatus::Write {
213            return Ok(TapeSkipOutcome::Stop(MachineStop::CopyCheck));
214        }
215        self.status = TapeStatus::Read;
216        self.rewound = false;
217        if self.position >= self.records.len() {
218            return Ok(TapeSkipOutcome::EndOfFileSkip);
219        }
220        self.position += 1;
221        Ok(TapeSkipOutcome::Ok)
222    }
223
224    /// Executes `SBTP*`: backspaces over the preceding unit record without reading.
225    pub fn skip_backward(&mut self) -> Result<TapeSkipOutcome, SpeedError> {
226        if self.status == TapeStatus::Write {
227            return Ok(TapeSkipOutcome::Stop(MachineStop::CopyCheck));
228        }
229        self.status = TapeStatus::Read;
230        self.rewound = false;
231        if self.position == 0 {
232            return Ok(TapeSkipOutcome::EndOfFileSkip);
233        }
234        self.position -= 1;
235        Ok(TapeSkipOutcome::Ok)
236    }
237
238    /// Injects a checksum discrepancy on a recorded unit record for testing.
239    pub fn inject_checksum_error(&mut self, record_index: usize) {
240        if let Some(rec) = self.records.get_mut(record_index) {
241            rec.checksum_corrupted = true;
242        }
243    }
244
245    /// Injects an even-parity redundancy error on a recorded unit record for testing.
246    pub fn inject_parity_error(&mut self, record_index: usize) {
247        if let Some(rec) = self.records.get_mut(record_index) {
248            rec.parity_corrupted = true;
249        }
250    }
251
252    /// Computes the 7th-channel odd-parity redundancy bit (`0` or `1`) for a 6-bit group.
253    pub fn compute_odd_parity_bit(six_bit_group: u8) -> u8 {
254        let ones = (six_bit_group & 0x3F).count_ones();
255        u8::from(ones.is_multiple_of(2))
256    }
257}

259/// Simulated 1024-word magnetic drum unit (`P` or `Q`).
260#[derive(Debug, Clone)]
261pub struct DrumUnit {
262    id: DrumId,
263    words: Vec<Word72>,
264    inject_verify_error: bool,
265}
266
267impl DrumUnit {
268    /// Creates a new zero-initialized 1024-word drum unit.
269    pub fn new(id: DrumId) -> Self {
270        Self {
271            id,
272            words: vec![Word72::default(); DRUM_CAPACITY],
273            inject_verify_error: false,
274        }
275    }
276
277    /// Returns the drum unit identifier (`P` or `Q`).
278    pub fn id(&self) -> DrumId {
279        self.id
280    }
281
282    /// Executes `WRDR*`: writes `words` starting at `start` (`0..=1023`) and verifies against drum.
283    pub fn write_block(
284        &mut self,
285        start: Address,
286        words: &[Word72],
287    ) -> Result<DrumWriteOutcome, SpeedError> {
288        let start_idx = check_drum_bounds(start, words.len())?;
289        self.words[start_idx..start_idx + words.len()].copy_from_slice(words);
290        if self.inject_verify_error {
291            self.inject_verify_error = false;
292            return Ok(DrumWriteOutcome::VerifyDiscrepancySkip);
293        }
294        Ok(DrumWriteOutcome::Ok)
295    }
296
297    /// Executes `RFDR*`: reads `count` consecutive words starting at `start` (`0..=1023`).
298    pub fn read_block(&self, start: Address, count: usize) -> Result<Vec<Word72>, SpeedError> {
299        let start_idx = check_drum_bounds(start, count)?;
300        Ok(self.words[start_idx..start_idx + count].to_vec())
301    }
302
303    /// Forces the next drum write verification read-back to report a discrepancy.
304    pub fn inject_next_write_verify_error(&mut self) {
305        self.inject_verify_error = true;
306    }
307}

309/// Formatted printer line with computed `100 * C + n` identification number and up to 5 decimal floats.
310#[derive(Debug, Clone, PartialEq, Eq)]
311pub struct PrinterLine {
312    pub id: u32,
313    pub values: Vec<DecimalFloat>,
314}
315
316/// Line printer receiving floating decimal blocks from `PRINT` (`OP1 = 580`).
317#[derive(Debug, Clone, Default)]
318pub struct LinePrinter {
319    lines: Vec<PrinterLine>,
320}
321
322impl LinePrinter {
323    /// Creates an empty printer output log.
324    pub fn new() -> Self {
325        Self::default()
326    }
327
328    /// Converts `values` to floating decimal (`|E| <= 236`) and prints 5 numbers per line.
329    pub fn print_block(
330        &mut self,
331        values: &[SpeedFloat],
332        c_id: u16,
333    ) -> Result<Vec<PrinterLine>, SpeedError> {
334        let decimals: Vec<DecimalFloat> = values
335            .iter()
336            .copied()
337            .map(DecimalFloat::from_speed_float)
338            .collect::<Result<_, _>>()?;
339        let printed: Vec<PrinterLine> = decimals
340            .chunks(PRINTER_VALUES_PER_LINE)
341            .enumerate()
342            .map(|(idx, chunk)| PrinterLine {
343                id: 100u32 * u32::from(c_id) + (idx as u32) + 1,
344                values: chunk.to_vec(),
345            })
346            .collect();
347        self.lines.extend(printed.iter().cloned());
348        Ok(printed)
349    }
350
351    /// Returns all printed lines accumulated so far.
352    pub fn lines(&self) -> &[PrinterLine] {
353        &self.lines
354    }
