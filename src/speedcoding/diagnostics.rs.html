10/// Active pass within a SpeedCo I checking loop (manual section 10).
11#[derive(Debug, Clone, Copy, PartialEq, Eq)]
12pub enum CheckingPass {
13    Inactive,
14    Pass1,
15    Pass2,
16}
17
18/// Result of executing `ECHTR` (`OP2 = 254`) at the end of a checking loop.
19#[derive(Debug, Clone, Copy, PartialEq, Eq)]
20pub enum EchTrOutcome {
21    RepeatAt(Address),
22    VerifiedSkipNext,
23    DiscrepancyContinue,
24}
25
26#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
27enum CheckingPhase {
28    #[default]
29    Unprepared,
30    ReadyForPass1,
31    InPass1,
32    ReadyForPass2,
33    InPass2,
34}
35
36/// State machine tracking `PRCH`, `STCH`, `ECHTR`, and fixed-point check counters 1 and 2.
37#[derive(Debug, Clone, Default)]
38pub struct CheckingState {
39    phase: CheckingPhase,
40    counter_1: u128,
41    counter_2: u128,
42}
43
44impl CheckingState {
45    /// Creates an unprepared checking state.
46    pub fn new() -> Self {
47        Self::default()
48    }
49
50    /// Returns the current checking pass state.
51    pub fn pass(&self) -> CheckingPass {
52        match self.phase {
53            CheckingPhase::InPass1 => CheckingPass::Pass1,
54            CheckingPhase::InPass2 => CheckingPass::Pass2,
55            CheckingPhase::Unprepared
56            | CheckingPhase::ReadyForPass1
57            | CheckingPhase::ReadyForPass2 => CheckingPass::Inactive,
58        }
59    }
60
61    /// Executes `PRCH` (`OP2 = 232`), initializing control state before `STCH`.
62    pub fn prepare_check(&mut self) {
63        self.phase = CheckingPhase::ReadyForPass1;
64        self.counter_1 = 0;
65        self.counter_2 = 0;
66    }
67
68    /// Executes `STCH` (`OP2 = 253`), entering pass 1 or pass 2 and recording `first_result`.
69    pub fn start_check(&mut self, first_result: Word72) -> Result<(), SpeedError> {
70        match self.phase {
71            CheckingPhase::ReadyForPass1 => {
72                self.counter_1 = first_result.raw();
73                self.counter_2 = 0;
74                self.phase = CheckingPhase::InPass1;
75                Ok(())
76            }
77            CheckingPhase::ReadyForPass2 => {
78                self.counter_2 = first_result.raw();
79                self.phase = CheckingPhase::InPass2;
80                Ok(())
81            }
82            CheckingPhase::Unprepared | CheckingPhase::InPass1 | CheckingPhase::InPass2 => Err(
83                SpeedError::CheckingProtocolViolation(
84                    "STCH requires prior PRCH or Pass-1 ECHTR".to_string(),
85                ),
86            ),
87        }
88    }
89
90    /// Accumulates an intermediate computed result into the active check counter (fixed-point sum).
91    pub fn record_computed_result(&mut self, result: Word72) {
92        match self.phase {
93            CheckingPhase::InPass1 => {
94                self.counter_1 = self.counter_1.wrapping_add(result.raw());
95            }
96            CheckingPhase::InPass2 => {
97                self.counter_2 = self.counter_2.wrapping_add(result.raw());
98            }
99            CheckingPhase::Unprepared
100            | CheckingPhase::ReadyForPass1
101            | CheckingPhase::ReadyForPass2 => {}
102        }
103    }
104
105    /// Executes `ECHTR` (`OP2 = 254`): loops to `target` after pass 1 or compares counters after pass 2.
106    pub fn end_check_and_transfer(
107        &mut self,
108        last_result: Word72,
109        target: Address,
110    ) -> Result<EchTrOutcome, SpeedError> {
111        match self.phase {
112            CheckingPhase::InPass1 => {
113                self.counter_1 = self.counter_1.wrapping_add(last_result.raw());
114                self.phase = CheckingPhase::ReadyForPass2;
115                Ok(EchTrOutcome::RepeatAt(target))
116            }
117            CheckingPhase::InPass2 => {
118                self.counter_2 = self.counter_2.wrapping_add(last_result.raw());
119                self.phase = CheckingPhase::ReadyForPass1;
120                if self.counter_1 == self.counter_2 {
121                    Ok(EchTrOutcome::VerifiedSkipNext)
122                } else {
123                    Ok(EchTrOutcome::DiscrepancyContinue)
124                }
125            }
126            CheckingPhase::Unprepared
127            | CheckingPhase::ReadyForPass1
128            | CheckingPhase::ReadyForPass2 => Err(SpeedError::CheckingProtocolViolation(
129                "ECHTR requires active STCH pass".to_string(),
130            )),
131        }
132    }
133
134    /// Returns `true` when executing the second pass of a checking loop.
135    pub fn is_second_pass(&self) -> bool {
136        self.phase == CheckingPhase::InPass2
137    }
138
139    pub(crate) fn is_awaiting_second_pass(&self) -> bool {
140        self.phase == CheckingPhase::ReadyForPass2
141    }
142
143    /// Returns the accumulated value in check counter 1.
144    pub fn counter_1(&self) -> u128 {
145        self.counter_1
146    }
147
148    /// Returns the accumulated value in check counter 2.
149    pub fn counter_2(&self) -> u128 {
150        self.counter_2
151    }
152}

154/// Operator panel listing sense switches labeled 1, 2, 3 (`L`-codes 0, 3, 6).
155#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
156pub struct ListingSwitches {
157    pub switch_1_l0: bool,
158    pub switch_2_l3: bool,
159    pub switch_3_l6: bool,

161
162impl ListingSwitches {
163    /// Returns `true` if the panel switch corresponding to `l_code` is ON.
164    pub fn should_list(self, l_code: LCode) -> bool {
165        match l_code.switch_number() {
166            1 => self.switch_1_l0,
167            2 => self.switch_2_l3,
168            3 => self.switch_3_l6,
169            other => unreachable!("validated LCode switch number out of range: {other}"),
170        }
171    }

174/// Formatted operand value printed in columns 14–16 of a listing line (manual page 11-2).

182/// Single 16-column SpeedCo I listing trace line (manual page 11-1).
183#[derive(Debug, Clone, PartialEq, Eq)]
184pub struct ListingEntry {
185    pub location: Address,
186    pub op1: Op1,
187    pub r_code: RCode,
188    pub a: Address,
189    pub b: Address,
190    pub c: Address,
191    pub op2: Op2,
192    pub d: Address,
193    pub l_code: LCode,
194    pub address_counter: i32,
195    pub r_a: u16,
196    pub r_b: u16,
197    pub r_c: u16,

199    pub b_val: ListedValue,
200    pub c_val: ListedValue,
201}
202
203fn format_single_listed_value(val: Option<SpeedFloat>) -> ListedValue {
204    let Some(sf) = val else {
205        return ListedValue::Blank;
206    };
207    match DecimalFloat::from_speed_float(sf) {
208        Ok(dec) => ListedValue::Decimal(dec),
209        Err(SpeedError::DecimalExponentOutOfRange(_)) => ListedValue::ExponentOverflow {
210            positive: !sf.exp_negative,
211        },
212        Err(_) => ListedValue::Blank,
213    }
214}

216/// Drum-backed 10-line listing buffer (manual pages 11-1..11-2).
217#[derive(Debug, Clone, Default)]
218pub struct ListingBuffer {
219    pending: Vec<ListingEntry>,
220    flushed: Vec<Vec<ListingEntry>>,
221}
222
223impl ListingBuffer {
224    /// Creates an empty listing buffer.
225    pub fn new() -> Self {
226        Self::default()
227    }
228
229    /// Records an entry when enabled by switches and not in checking pass 2, flushing every 10 lines.
230    pub fn record(
231        &mut self,
232        entry: ListingEntry,
233        switches: ListingSwitches,
234        is_check_pass_2: bool,
235    ) -> Option<Vec<ListingEntry>> {
236        if is_check_pass_2 || !switches.should_list(entry.l_code) {
237            return None;
238        }
239        self.pending.push(entry);
240        if self.pending.len() >= LISTING_BLOCK_SIZE {
241            let block = std::mem::take(&mut self.pending);
242            self.flushed.push(block.clone());
243            Some(block)
244        } else {
245            None
246        }
247    }
248
249    /// Returns number of buffered lines currently awaiting a 10-line flush.
250    pub fn pending_count(&self) -> usize {
251        self.pending.len()
252    }
253
254    /// Returns all flushed 10-line listing blocks.
255    pub fn flushed_blocks(&self) -> &[Vec<ListingEntry>] {
256        &self.flushed
257    }
258
259    /// Formats `Q(A)`, `Q(B)`, `Q(C)` for listing, applying manual page 11-1/11-2 sign/overflow rules.
260    pub fn format_listed_operands(
261        op1: Op1,
262        qa: Option<SpeedFloat>,
263        qb: Option<SpeedFloat>,
264        qc: Option<SpeedFloat>,
265    ) -> (ListedValue, ListedValue, ListedValue) {
266        let (eff_qa, eff_qb) = match op1 {
267            Op1::Sub => (qa, qb.map(|v| -v)),
268            Op1::AddAb => (qa, qb.map(SpeedFloat::abs)),
269            Op1::AbAdd => (qa.map(SpeedFloat::abs), qb.map(SpeedFloat::abs)),
270            Op1::SubAb => (qa, qb.map(|v| -v.abs())),
271            Op1::AbSub => (qa.map(SpeedFloat::abs), qb.map(|v| -v.abs())),
272            Op1::NgMpy | Op1::NgDiv => (qa.map(|v| -v), qb),
273            _ => (qa, qb),
274        };
275        (
276            format_single_listed_value(eff_qa),
277            format_single_listed_value(eff_qb),
278            format_single_listed_value(qc),
279        )
280    }
281}

309/// Analytical timing model from manual Section 12 and Appendices 1–2.

312impl TimingCalculator {
313    /// Returns average and maximum execution time (ms) for `OP1`.
314    pub fn op1_time_ms(op1: Op1, ctx: TimingContext) -> TimingRange {
315        let n = ctx.word_count as f64;
316        match op1 {
317            Op1::Add | Op1::Sub | Op1::AddAb | Op1::AbAdd | Op1::SubAb | Op1::AbSub => {
318                TimingRange {
319                    avg_ms: 4.200,
320                    max_ms: 19.272,
321                }
322            }
323            Op1::Mpy | Op1::NgMpy => TimingRange {
324                avg_ms: 3.546,
325                max_ms: 3.876,
326            },
327            Op1::Div | Op1::NgDiv => TimingRange {
328                avg_ms: 4.236,
329                max_ms: 4.440,
330            },
331            Op1::NoOp => TimingRange {
332                avg_ms: 2.736,
333                max_ms: 2.736,
334            },
335            Op1::Sqrt => TimingRange {
336                avg_ms: 8.200,
337                max_ms: 9.016,
338            },
339            Op1::Sine => TimingRange {
340                avg_ms: 20.0,
341                max_ms: 26.0,
342            },
343            Op1::Artan => TimingRange {
344                avg_ms: 24.0,
345                max_ms: 32.0,
346            },
347            Op1::Exp => TimingRange {
348                avg_ms: 18.0,
349                max_ms: 24.0,
350            },
351            Op1::Ln => TimingRange {
352                avg_ms: 33.0,
353                max_ms: 42.0,
354            },
355            Op1::WrTpJ | Op1::WrTpK | Op1::WrTpL | Op1::WrTpM => {
356                let cond = ctx.tape_condition.unwrap_or(TapeCondition::Cond1AfterWrite);
357                let base_avg = match cond {
358                    TapeCondition::Cond1AfterWrite => 14.000,
359                    TapeCondition::Cond2AfterRewind
360                    | TapeCondition::Cond3AfterReadOrSkipFwd
361                    | TapeCondition::Cond4AfterReadOrSkipBwd
362                    | TapeCondition::Cond5AfterEndFile => 19.000,
363                };
364                TimingRange {
365                    avg_ms: base_avg + 1.600 * n,
366                    max_ms: base_avg + 5.000 + 1.600 * n,
367                }
368            }
369            Op1::RfTpJ
370            | Op1::RfTpK
371            | Op1::RfTpL
372            | Op1::RfTpM
373            | Op1::SfTpJ
374            | Op1::SfTpK
375            | Op1::SfTpL
376            | Op1::SfTpM
377            | Op1::SbTpJ
378            | Op1::SbTpK
379            | Op1::SbTpL
380            | Op1::SbTpM => TimingRange {
381                avg_ms: 14.000 + 1.600 * n,
382                max_ms: 19.000 + 1.600 * n,
383            },
384            Op1::RbTpJ | Op1::RbTpK | Op1::RbTpL | Op1::RbTpM => TimingRange {
385                avg_ms: 42.000 + 4.800 * n,
386                max_ms: 57.000 + 4.800 * n,
387            },
388            Op1::RwTpJ | Op1::RwTpK | Op1::RwTpL | Op1::RwTpM => TimingRange {
389                avg_ms: 3.500,
390                max_ms: 3.500,
391            },
392            Op1::EfTpJ | Op1::EfTpK | Op1::EfTpL | Op1::EfTpM => TimingRange {
393                avg_ms: 960.0,
394                max_ms: 960.0,
395            },
396            Op1::WrDrP | Op1::WrDrQ => TimingRange {
397                avg_ms: 20.0 + 0.096 * n,
398                max_ms: 30.0 + 0.096 * n,
399            },
400            Op1::RfDrP | Op1::RfDrQ => TimingRange {
401                avg_ms: 10.0 + 0.048 * n,
402                max_ms: 20.0 + 0.048 * n,
403            },
404            Op1::Print => {
405                if ctx.is_check_pass_2 {
406                    TimingRange {
407                        avg_ms: 2.896,
408                        max_ms: 2.896,
409                    }
410                } else {
411                    let lines = ctx.word_count.div_ceil(PRINTER_VALUES_PER_LINE) as f64;
412                    TimingRange {
413                        avg_ms: 1663.0 + 400.0 * lines,
414                        max_ms: 1703.0 + 400.0 * lines,
415                    }
416                }
417            }
418        }
419    }

421    /// Returns execution time (ms) for `OP2` under the given branch/condition outcome.
422    pub fn op2_time_ms(op2: Op2, condition_met: bool) -> f64 {
423        match op2 {
424            Op2::NoOp => 0.000,
425            Op2::Tr => 0.768,
426            Op2::TrPl | Op2::TrMn | Op2::TrZ | Op2::SnTrP | Op2::SnTrQ => {
427                if condition_met { 0.876 } else { 0.648 }
428            }
429            Op2::TiA
430            | Op2::TiB
431            | Op2::TiC
432            | Op2::TiAB
433            | Op2::TiBC
434            | Op2::TiAC
435            | Op2::TiABC => 2.256,
436            Op2::TdA
437            | Op2::TdB
438            | Op2::TdC
439            | Op2::TdAB
440            | Op2::TdBC
441            | Op2::TdAC
442            | Op2::TdABC => 3.036,
443            Op2::SetRA | Op2::SetRB | Op2::SetRC | Op2::PrCh | Op2::StCh => 1.200,
444            Op2::SkRA | Op2::SkRB | Op2::SkRC | Op2::Skip | Op2::EchTr => {
445                if condition_met { 1.368 } else { 1.260 }
446            }
447            Op2::RAddA
448            | Op2::RAddB
449            | Op2::RAddC
450            | Op2::RAddD
451            | Op2::AddA
452            | Op2::AddB
453            | Op2::AddC
454            | Op2::AddD
455            | Op2::SubA
456            | Op2::SubB
457            | Op2::SubC
458            | Op2::SubD
459            | Op2::StA
460            | Op2::StB
461            | Op2::StC
462            | Op2::StD => 1.536,
463            Op2::Stop => 0.816,
464        }
465    }

467    /// Returns total instruction time (ms) = `op1_time_ms + op2_time_ms`.
468    pub fn instruction_time_ms(op1: Op1, op2: Op2, ctx: TimingContext) -> TimingRange {
469        let op1_range = Self::op1_time_ms(op1, ctx);
470        let op2_ms = Self::op2_time_ms(op2, ctx.op2_condition_met);
471        TimingRange {
472            avg_ms: op1_range.avg_ms + op2_ms,
473            max_ms: op1_range.max_ms + op2_ms,
474        }
475    }

477    /// Computes total physical tape length in inches (manual page 12-3).
478    pub fn tape_file_length_inches(record_word_counts: &[usize]) -> f64 {
479        let k = record_word_counts.len();
480        let data_inches: f64 = record_word_counts
481            .iter()
482            .map(|&n| (n as f64 + 0.5) * 3.0 / 25.0)
483            .sum();
484        2.0 * 72.0 + data_inches + (k.saturating_sub(1) as f64)
485    }
486
487    /// Computes tape rewind time in seconds (`length_inches / 75.0`).
488    pub fn tape_rewind_time_secs(record_word_counts: &[usize]) -> f64 {
489        Self::tape_file_length_inches(record_word_counts) / 75.0
490    }
491
492    /// Returns time in seconds required to write a 72-inch end-of-file gap (`0.96 s`).
493    pub fn tape_eof_gap_time_secs() -> f64 {
494        72.0 / 75.0
495    }

497    /// Computes overlap delay `(1.600 * n - t).max(0.0)` in ms per Note 1 (page 12-2).
498    pub fn tape_skip_delay_ms(record_words: usize, elapsed_ms: f64) -> f64 {
499        (1.600 * (record_words as f64) - elapsed_ms).max(0.0)
500    }

502    /// Approximates card loading time `(0.4)(n) + (0.6)(c) + e` in seconds (page 12-5).
503    pub fn loading_time_secs(
504        total_cards: usize,
505        control_cards: usize,
506        speedco_on_drums: bool,
507    ) -> f64 {
508        let e = if speedco_on_drums { 2.0 } else { 55.0 };
509        0.4 * (total_cards as f64) + 0.6 * (control_cards as f64) + e
510    }
