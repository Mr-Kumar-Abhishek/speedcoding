52/// 10-bit electrostatic or drum address (`0..=1023`), with user storage in `300..=1013`.
53#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
54pub struct Address {
55    raw: u16,
56}
57
58impl Address {
59    /// Creates an address within the 1024-word address space (`0..=1023`).
60    pub fn new(raw: u16) -> Result<Self, SpeedError> {
61        if raw < TOTAL_MEM_WORDS as u16 {
62            Ok(Self { raw })
63        } else {
64            Err(SpeedError::AddressOutOfBounds(raw))
65        }
66    }
67
68    /// Creates a user-accessible electrostatic storage address (`300..=1013`).
69    pub fn user(raw: u16) -> Result<Self, SpeedError> {
70        let addr = Self::new(raw)?;
71        if addr.is_user_accessible() {
72            Ok(addr)
73        } else {
74            Err(SpeedError::ControlRegionAccess(raw))
75        }
76    }
77
78    /// Returns the raw integer address value (`0..=1023`).
79    pub fn raw(self) -> u16 {
80        self.raw
81    }
82
83    /// Returns `true` when the address lies inside user storage (`300..=1013`).
84    pub fn is_user_accessible(self) -> bool {
85        (USER_MEM_START..=USER_MEM_END).contains(&self.raw)
86    }
87}

89/// 72-bit IBM 701 word stored right-aligned in the lower 72 bits of `u128`.
90#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
91pub struct Word72 {
92    bits: u128,
93}
94
95impl Word72 {
96    /// Validates that only bits `0..72` are populated and wraps them in `Word72`.
97    pub fn new(bits: u128) -> Result<Self, SpeedError> {
98        if (bits & !WORD72_MASK) != 0 {
99            Err(SpeedError::InvalidWordBits(bits))
100        } else {
101            Ok(Self { bits })
102        }
103    }
104
105    /// Returns the raw 72-bit value right-aligned in `u128`.
106    pub fn raw(self) -> u128 {
107        self.bits
108    }
109
110    /// Extracts a single bit using IBM 701 1-based MSB numbering (`1..=72`).
111    pub fn bit(self, pos_1based: u8) -> bool {
112        assert!(
113            (1..=WORD_BITS as u8).contains(&pos_1based),
114            "bit position {pos_1based} out of 1..=72"
115        );
116        ((self.bits >> (WORD_BITS as u8 - pos_1based)) & 1) != 0
117    }
118
119    /// Extracts an inclusive bit slice using IBM 701 1-based MSB numbering (`1..=72`).
120    pub fn bits(self, start_1based: u8, end_1based: u8) -> u64 {
121        assert!(
122            (1..=WORD_BITS as u8).contains(&start_1based)
123                && (start_1based..=WORD_BITS as u8).contains(&end_1based),
124            "invalid 1-based bit slice {start_1based}..={end_1based}"
125        );
126        let width = (end_1based - start_1based + 1) as u32;
127        assert!(width <= 64, "bit slice width {width} exceeds u64");
128        let mask = if width == 64 {
129            u64::MAX as u128
130        } else {
131            (1u128 << width) - 1
132        };
133        ((self.bits >> (WORD_BITS as u8 - end_1based)) & mask) as u64
134    }
135
136    /// Splits the 72-bit word into 12 six-bit tape groups (group 1 = bits `1..=6` .. group 12 = bits `67..=72`).
137    pub fn to_six_bit_groups(self) -> [u8; 12] {
138        let mut groups = [0u8; 12];
139        for (idx, slot) in groups.iter_mut().enumerate() {
140            *slot = ((self.bits >> (66 - 6 * idx)) & 0x3F) as u8;
141        }
142        groups
143    }
144
145    /// Reassembles a 72-bit word from 12 six-bit tape groups.
146    pub fn from_six_bit_groups(groups: [u8; 12]) -> Self {
147        let mut bits = 0u128;
148        for (idx, &group) in groups.iter().enumerate() {
149            bits |= ((group & 0x3F) as u128) << (66 - 6 * idx);
150        }
151        Self { bits }
152    }
153}

155/// Internal SpeedCo I floating binary representation (`F * 2^E`).
156#[derive(Debug, Clone, Copy, PartialEq, Eq)]
157pub struct SpeedFloat {
158    pub negative: bool,
159    pub fraction: u64,
160    pub exp_negative: bool,
161    pub exponent: u32,
162}
163
164impl SpeedFloat {
165    /// Returns floating-point zero (`F = 0`, `E = 0`).
166    pub fn zero() -> Self {
167        Self {
168            negative: false,
169            fraction: 0,
170            exp_negative: false,
171            exponent: 0,
172        }
173    }
174
175    /// Converts an IEEE-754 `f64` into normalized SpeedCo I 35-bit floating binary.
176    pub fn from_f64(value: f64) -> Result<Self, SpeedError> {
177        if !value.is_finite() {
178            return Err(SpeedError::BinaryExponentOutOfRange(i32::MAX));
179        }
180        if value == 0.0 {
181            return Ok(Self::zero());
182        }
183
184        let bits = value.to_bits();
185        let negative = (bits >> 63) != 0;
186        let raw_exp = ((bits >> 52) & 0x7FF) as i32;
187        let raw_mant = bits & ((1u64 << 52) - 1);
188
189        let (sig53, mut exp) = if raw_exp == 0 {
190            // Normalize subnormal mantissa so MSB sits at bit 52 (53-bit significand).
191            let shift = raw_mant.leading_zeros() - 11;
192            (raw_mant << shift, (1 - 1022) - (shift as i32))
193        } else {
194            ((1u64 << 52) | raw_mant, raw_exp - 1022)
195        };
196
197        // Round-to-nearest-even when truncating 53-bit significand down to 35 fraction bits.
198        const DROP_BITS: u32 = 53 - FRACTION_BITS;
199        let dropped = sig53 & ((1u64 << DROP_BITS) - 1);
200        let halfway = 1u64 << (DROP_BITS - 1);
201        let mut fraction = sig53 >> DROP_BITS;
202        if dropped > halfway || (dropped == halfway && (fraction & 1) != 0) {
203            fraction += 1;
204            if fraction == (1u64 << FRACTION_BITS) {
205                fraction >>= 1;
206                exp += 1;
207            }
208        }
209
210        if exp.unsigned_abs() > MAX_BINARY_EXPONENT {
211            return Err(SpeedError::BinaryExponentOutOfRange(exp));
212        }
213
214        Ok(Self {
215            negative,
216            fraction,
217            exp_negative: exp < 0,
218            exponent: exp.unsigned_abs(),
219        })
220    }
221
222    /// Converts this SpeedCo I floating binary value to `f64`.
223    pub fn to_f64(self) -> f64 {
224        if self.fraction == 0 {
225            return 0.0;
226        }
227        let frac = (self.fraction as f64) / ((1u64 << FRACTION_BITS) as f64);
228        let mag = frac * 2.0_f64.powi(self.signed_exponent());
229        if self.negative { -mag } else { mag }
230    }
231
232    /// Returns the signed binary exponent (`-131_071..=131_071`).
233    pub fn signed_exponent(self) -> i32 {
234        if self.exp_negative {
235            -(self.exponent as i32)
236        } else {
237            self.exponent as i32
238        }
239    }
240
241    /// Returns `true` when the fractional part is zero.
242    pub fn is_zero_fraction(self) -> bool {
243        self.fraction == 0
244    }
245
246    /// Returns `true` if `fraction == 0` or `0.5 <= |F| < 1` (`fraction >= 2^34`).
247    pub fn is_normalized(self) -> bool {
248        self.fraction == 0 || (MIN_NORM_FRACTION..=FRACTION_MASK).contains(&self.fraction)
249    }
250
251    /// Returns `|self|`.
252    pub fn abs(self) -> Self {
253        Self {
254            negative: false,
255            ..self
256        }
257    }
258
259    /// Returns `-self`.
260    pub fn neg(self) -> Self {
261        Self {
262            negative: !self.negative,
263            ..self
264        }
265    }
266
267    /// Encodes into a 72-bit word (bit 1 sign, 2..=36 frac, 37 exp sign, 38..=54 exp, 55..=72 zero).
268    pub fn to_word(self) -> Word72 {
269        let mut bits = 0u128;
270        if self.negative {
271            bits |= 1u128 << 71;
272        }
273        bits |= ((self.fraction & FRACTION_MASK) as u128) << 36;
274        if self.exp_negative {
275            bits |= 1u128 << 35;
276        }
277        bits |= ((self.exponent & MAX_BINARY_EXPONENT) as u128) << 18;
278        Word72 { bits }
279    }
280
281    /// Decodes a 72-bit word into `SpeedFloat`.
282    pub fn from_word(word: Word72) -> Self {
283        Self {
284            negative: word.bit(1),
285            fraction: word.bits(2, 36),
286            exp_negative: word.bit(37),
287            exponent: word.bits(38, 54) as u32,
288        }
289    }
290}

292/// External SpeedCo I floating decimal representation (`F * 10^E`, 10-digit fraction, `|E| <= 236`).
293#[derive(Debug, Clone, Copy, PartialEq, Eq)]
294pub struct DecimalFloat {
295    pub negative: bool,
296    pub digits: u64,
297    pub exponent: i16,
298}
299
300impl DecimalFloat {
301    /// Constructs and validates a floating decimal value (`|exponent| <= 236`).
302    pub fn new(negative: bool, digits: u64, exponent: i16) -> Result<Self, SpeedError> {
303        if !(-MAX_DECIMAL_EXPONENT..=MAX_DECIMAL_EXPONENT).contains(&exponent) {
304            return Err(SpeedError::DecimalExponentOutOfRange(exponent as i32));
305        }
306        if digits >= TEN_DIGITS_LIMIT {
307            return Err(SpeedError::InvalidCardFormat(format!(
308                "decimal fraction exceeds 10 digits: {digits}"
309            )));
310        }
311        Ok(Self {
312            negative,
313            digits,
314            exponent,
315        })
316    }
317
318    /// Converts external decimal floating point to internal binary `SpeedFloat`.
319    pub fn to_speed_float(self) -> Result<SpeedFloat, SpeedError> {
320        if self.digits == 0 {
321            return Ok(SpeedFloat::zero());
322        }
323        let sign = if self.negative { "-" } else { "" };
324        let repr = format!("{sign}0.{:010}e{}", self.digits, self.exponent);
325        let val: f64 = repr.parse().map_err(|err| {
326            SpeedError::InvalidCardFormat(format!("invalid decimal float '{repr}': {err}"))
327        })?;
328        SpeedFloat::from_f64(val)
329    }
330
331    /// Converts internal binary `SpeedFloat` to external decimal `DecimalFloat`, rejecting `|E| > 236`.
332    pub fn from_speed_float(val: SpeedFloat) -> Result<Self, SpeedError> {
333        if val.is_zero_fraction() {
334            return Ok(Self {
335                negative: false,
336                digits: 0,
337                exponent: 0,
338            });
339        }
340
341        // Reject out-of-range decimal exponents before calling to_f64() so large binary exponents do not overflow f64.
342        let frac = (val.fraction as f64) / ((1u64 << FRACTION_BITS) as f64);
343        let log10_val = frac.log10() + (val.signed_exponent() as f64) * std::f64::consts::LOG10_2;
344        let est_dec_exp = log10_val.floor() as i32 + 1;
345        let dec_limit = MAX_DECIMAL_EXPONENT as i32;
346        if est_dec_exp < -dec_limit - 1 || est_dec_exp > dec_limit + 1 {
347            return Err(SpeedError::DecimalExponentOutOfRange(est_dec_exp));
348        }
349
350        let formatted = format!("{:.9e}", val.to_f64().abs());
351        let (mant_str, exp_str) = formatted.split_once('e').ok_or_else(|| {
352            SpeedError::ParseError(format!("unexpected scientific format: {formatted}"))
353        })?;
354        let digits_str: String = mant_str.chars().filter(|&ch| ch != '.').collect();
355        let digits: u64 = digits_str.parse().map_err(|err| {
356            SpeedError::ParseError(format!("invalid decimal digits '{digits_str}': {err}"))
357        })?;
358        let sci_exp: i32 = exp_str.parse().map_err(|err| {
359            SpeedError::ParseError(format!("invalid decimal exponent '{exp_str}': {err}"))
360        })?;
361        let dec_exp = sci_exp + 1;
362        if !(-dec_limit..=dec_limit).contains(&dec_exp) {
363            return Err(SpeedError::DecimalExponentOutOfRange(dec_exp));
364        }
365
366        Ok(Self {
367            negative: val.negative,
368            digits,
369            exponent: dec_exp as i16,
370        })
371    }
372}

374/// 3-bit address modification selector (`0..=7`) controlling `R_A`, `R_B`, and `R_C`.
375#[derive(Debug, Clone, Copy, PartialEq, Eq)]
376pub struct RCode {
377    raw: u8,
378}
379
380impl RCode {
381    /// Validates `val <= 7`.
382    pub fn new(val: u8) -> Result<Self, SpeedError> {
383        if val <= 7 {
384            Ok(Self { raw: val })
385        } else {
386            Err(SpeedError::InvalidRCode(val))
387        }
388    }
389
390    /// Returns raw R-code value (`0..=7`).
391    pub fn raw(self) -> u8 {
392        self.raw
393    }
394
395    /// Returns `true` when bit 4 is set (`R` in `{4, 5, 6, 7}`), adding `R_A` to `A`.
396    pub fn modifies_a(self) -> bool {
397        (self.raw & 0b100) != 0
398    }
399
400    /// Returns `true` when bit 2 is set (`R` in `{2, 3, 6, 7}`), adding `R_B` to `B`.
401    pub fn modifies_b(self) -> bool {
402        (self.raw & 0b010) != 0
403    }
404
405    /// Returns `true` when bit 1 is set (`R` in `{1, 3, 5, 7}`), adding `R_C` to `C`.
406    pub fn modifies_c(self) -> bool {
407        (self.raw & 0b001) != 0
408    }
409}

411/// Selective listing code (`0`, `3`, or `6`), corresponding to panel switches 1, 2, or 3.
412#[derive(Debug, Clone, Copy, PartialEq, Eq)]
413pub struct LCode {
414    raw: u8,
415}
416
417impl LCode {
418    /// Validates `val` is one of `0`, `3`, or `6`.
419    pub fn new(val: u8) -> Result<Self, SpeedError> {
420        match val {
421            0 | 3 | 6 => Ok(Self { raw: val }),
422            _ => Err(SpeedError::InvalidLCode(val)),
423        }
424    }
425
426    /// Returns raw L-code (`0`, `3`, or `6`).
427    pub fn raw(self) -> u8 {
428        self.raw
429    }
430
431    /// Maps `0 -> 1`, `3 -> 2`, `6 -> 3`.
432    pub fn switch_number(self) -> u8 {
433        match self.raw {
434            0 => 1,
435            3 => 2,
436            6 => 3,
437            other => unreachable!("validated LCode invariant broken: {other}"),
438        }
439    }
440}

700impl Op2 {
701    /// Returns the 3-digit numerical operation code for `OP2`.
702    pub fn code(self) -> u16 {
703        OP2_TABLE
704            .iter()
705            .find(|&&(op, _, _)| op == self)
706            .expect("every Op2 variant exists in OP2_TABLE")
707            .2
708    }
709
710    /// Returns the alphabetical mnemonic for `OP2` (`""` for `NoOp`).
711    pub fn mnemonic(self) -> &'static str {
712        OP2_TABLE
713            .iter()
714            .find(|&&(op, _, _)| op == self)
715            .expect("every Op2 variant exists in OP2_TABLE")
716            .1
717    }
718
719    /// Decodes a 3-digit numerical `OP2` code (`0` -> `Op2::NoOp`).
720    pub fn from_code(code: u16) -> Result<Self, SpeedError> {
721        OP2_TABLE
722            .iter()
723            .find(|&&(_, _, c)| c == code)
724            .map(|&(op, _, _)| op)
725            .ok_or(SpeedError::InvalidOp2Code(code))
726    }
727
728    /// Decodes an alphabetical `OP2` mnemonic (`""` or `"-"` -> `Op2::NoOp`).
729    pub fn from_mnemonic(s: &str) -> Result<Self, SpeedError> {
730        let trimmed = s.trim();
731        if trimmed.is_empty() || trimmed == "-" {
732            return Ok(Op2::NoOp);
733        }
734        OP2_TABLE
735            .iter()
736            .find(|&&(_, mnem, _)| mnem == trimmed)
737            .map(|&(op, _, _)| op)
738            .ok_or_else(|| SpeedError::InvalidMnemonic(s.to_string()))
739    }

742/// Decoded SpeedCo I instruction occupying one 72-bit word (manual page 4-4).
743#[derive(Debug, Clone, Copy, PartialEq, Eq)]
744pub struct Instruction {
745    pub r_code: RCode,
746    pub op1: Op1,
747    pub op2: Op2,
748    pub d: Address,
749    pub a: Address,
750    pub b: Address,
751    pub c: Address,
752    pub l_code: LCode,
753}
754
755impl Instruction {
756    /// Encodes the instruction into its 72-bit memory layout (page 4-4).
757    pub fn to_word(self) -> Word72 {
758        let bits = ((self.r_code.raw() as u128) << 67)
759            | ((self.op1.code() as u128) << 54)
760            | ((self.op2.code() as u128) << 46)
761            | ((self.d.raw() as u128) << 36)
762            | ((self.a.raw() as u128) << 25)
763            | ((self.b.raw() as u128) << 15)
764            | ((self.l_code.raw() as u128) << 10)
765            | (self.c.raw() as u128);
766        Word72 { bits }
767    }
768
769    /// Decodes a 72-bit instruction word and validates field constraints.
770    pub fn from_word(word: Word72) -> Result<Self, SpeedError> {
771        if word.bit(2) || word.bit(6) || word.bit(37) || word.bits(58, 59) != 0 {
772            return Err(SpeedError::InvalidWordBits(word.raw()));
773        }
774        let r_code = RCode::new(word.bits(3, 5) as u8)?;
775        let op1 = Op1::from_code(word.bits(7, 18) as u16)?;
776        let op2 = Op2::from_code(word.bits(19, 26) as u16)?;
777        if op1.forbids_op2() && op2 != Op2::NoOp {
778            return Err(SpeedError::ForbiddenOp2WithSkipOp1 { op1, op2 });
779        }
780        let d = Address::new(word.bits(27, 36) as u16)?;
781        let a = Address::new(word.bits(38, 47) as u16)?;
782        let b = Address::new(word.bits(48, 57) as u16)?;
783        let l_code = LCode::new(word.bits(60, 62) as u8)?;
784        let c = Address::new(word.bits(63, 72) as u16)?;
785
786        Ok(Self {
787            r_code,
788            op1,
789            op2,
790            d,
791            a,
792            b,
793            c,
794            l_code,
795        })
796    }
797}

799/// 1024-word electrostatic cathode-ray-tube memory with user-accessible cells `300..=1013`.
800#[derive(Debug, Clone)]
801pub struct ElectrostaticMemory {
802    words: Vec<Word72>,
803}
804
805impl Default for ElectrostaticMemory {
806    fn default() -> Self {
807        Self::new()
808    }
809}
810
811impl ElectrostaticMemory {
812    /// Creates zero-initialized electrostatic memory.
813    pub fn new() -> Self {
814        Self {
815            words: vec![Word72::default(); TOTAL_MEM_WORDS],
816        }
817    }
818
819    /// Reads a raw 72-bit word from user storage (`300..=1013`), rejecting control cells.
820    pub fn read(&self, addr: Address) -> Result<Word72, SpeedError> {
821        if !addr.is_user_accessible() {
822            return Err(SpeedError::ControlRegionAccess(addr.raw()));
823        }
824        Ok(self.words[addr.raw() as usize])
825    }
826
827    /// Writes a raw 72-bit word into user storage (`300..=1013`), rejecting control cells.
828    pub fn write(&mut self, addr: Address, word: Word72) -> Result<(), SpeedError> {
829        if !addr.is_user_accessible() {
830            return Err(SpeedError::ControlRegionAccess(addr.raw()));
831        }
832        self.words[addr.raw() as usize] = word;
833        Ok(())
834    }
835
836    /// Reads and decodes a `SpeedFloat` from user storage (`300..=1013`).
837    pub fn read_float(&self, addr: Address) -> Result<SpeedFloat, SpeedError> {
838        Ok(SpeedFloat::from_word(self.read(addr)?))
839    }
840
841    /// Encodes and writes a `SpeedFloat` into user storage (`300..=1013`).
842    pub fn write_float(&mut self, addr: Address, val: SpeedFloat) -> Result<(), SpeedError> {
843        self.write(addr, val.to_word())
844    }
845
846    /// Reads and decodes an `Instruction` from user storage (`300..=1013`).
847    pub fn read_instruction(&self, addr: Address) -> Result<Instruction, SpeedError> {
848        Instruction::from_word(self.read(addr)?)
849    }
850
851    /// Encodes and writes an `Instruction` into user storage (`300..=1013`).
852    pub fn write_instruction(&mut self, addr: Address, instr: Instruction) -> Result<(), SpeedError> {
853        self.write(addr, instr.to_word())
854    }
855}
